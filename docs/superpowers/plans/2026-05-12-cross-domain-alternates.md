## Scope Decisions Locked Here

These are the judgment calls made up front; they become entries in `docs/DECISIONS.md` at the matching implementation step.

- **Per-domain self-contained scripts** rather than a shared `alternates_engine.py`. Matches existing `BM_GP.py` / `BM_TabPFN.py` / Callum's TBCM-script convention. Cost: ~80 lines of harness duplicated per domain. Benefit: each script is independently runnable and readable.
- **Heatmap "ground truth" strategy:** scattered FEM (BM) / TBCM / female-BCM dataset points overlaid on each method's predicted surface, at a fixed `PS` (median of the dataset), sweeping `(a_CT, a_TA)`. We do **not** re-run FEM on a grid (~8 min × 2500 cells = infeasible) and we do **not** designate any method as "ground truth." Misalignment of dots vs surface = method fails to capture trend.
- **Three-way comparison per domain (head-to-head architecture):** Each cross-domain panel plots GP, TabPFN, and the existing repo's best transfer method for that target. Alternates scripts (Tasks 2–5) only produce the GP/TabPFN side; the transfer side is read from existing artifacts:
  - **BM:** `Beam_Membrane/BM_Showcase.py`'s hardcoded `SMALL_N_TRANSFER` dict (means only — TODO #13) + `rf_transfer_results.json` for full-N reference line.

Sample sizes already at [25, 50, 100, 250, 500, 843]. **RF-only** comparator per 2026-05-12 decision — not NN, not PR.
- **N values:** `[5, 10, 20, 30, 50, 75, 100, 150, 200, 300, 500]` (matches existing BM alternates). 10 bootstrap replicates per N.

```python
"""
BCM -> TBCM: Gaussian Process baseline (non-transfer)

Mirrors Beam_Membrane/BM_GP.py exactly; only the data loader and output
path change. Tracks team/TODO.md cross-domain extension (May 2026 sync).

Output: writes 'GP' key into TBCM/results/alternates_results.json.
"""

import json
import os
import warnings
warnings.filterwarnings('ignore')

import numpy as np
import pandas as pd
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern, WhiteKernel, ConstantKernel as C
from sklearn.metrics import r2_score
from sklearn.preprocessing import StandardScaler

script_dir = os.path.dirname(os.path.abspath(__file__))

N_TARGETS = [5, 10, 20, 30, 50, 75, 100, 150, 200, 300, 500]
N_RUNS = 10

SHARED_FEATURES = ['a_CT', 'a_TA', 'PS']
TARGETS = ['F0', 'SPL']
TEST_POOL_SIZE = 1000


# ==================== METHOD ====================
def fit_predict_gp(X_train_sc, Y_train_sc, X_test_sc, scaler_Y_per_output):
    preds = np.zeros((X_test_sc.shape[0], Y_train_sc.shape[1]))
    for i in range(Y_train_sc.shape[1]):
        kernel = (
            C(1.0, (1e-3, 1e3))
            * Matern(length_scale=1.0, length_scale_bounds=(1e-2, 1e2), nu=2.5)
            + WhiteKernel(noise_level=1e-2, noise_level_bounds=(1e-5, 1e1))
        )
        gp = GaussianProcessRegressor(
            kernel=kernel, normalize_y=False,
            n_restarts_optimizer=3, random_state=42,
        )
        gp.fit(X_train_sc, Y_train_sc[:, i])
        pred_sc = gp.predict(X_test_sc)
        preds[:, i] = scaler_Y_per_output[i].inverse_transform(
            pred_sc.reshape(-1, 1)
        ).ravel()
    return preds


# ==================== HARNESS ====================
def run_single(n_target, df, random_state):
    if n_target >= len(df):
        df_train = df.copy()
    else:
        df_train = df.sample(n=n_target, random_state=random_state)
    test_pool = df.drop(df_train.index)
    if len(test_pool) < 100:
        test_pool = df
    df_test = test_pool.sample(
        n=min(TEST_POOL_SIZE, len(test_pool)), random_state=random_state
    )

    X_train = df_train[SHARED_FEATURES].values
    Y_train = df_train[TARGETS].values
    X_test = df_test[SHARED_FEATURES].values
    Y_test = df_test[TARGETS].values

    scaler_X = StandardScaler()
    X_train_sc = scaler_X.fit_transform(X_train)
    X_test_sc = scaler_X.transform(X_test)

    scalers_Y = [StandardScaler(), StandardScaler()]
    Y_train_sc = np.zeros_like(Y_train, dtype=float)
    for i in range(Y_train.shape[1]):
        Y_train_sc[:, i] = scalers_Y[i].fit_transform(
            Y_train[:, i].reshape(-1, 1)
        ).ravel()

    preds = fit_predict_gp(X_train_sc, Y_train_sc, X_test_sc, scalers_Y)
    return {
        'r2_f0': float(r2_score(Y_test[:, 0], preds[:, 0])),
        'r2_spl': float(r2_score(Y_test[:, 1], preds[:, 1])),
        'n_train': int(len(X_train)),
    }


def _existing_complete_ns(method_name='GP'):
    out_path = os.path.join(script_dir, 'results', 'alternates_results.json')
    if not os.path.exists(out_path):
        return set()
    with open(out_path) as f:
        data = json.load(f)
    block = data.get(method_name, {})
    done = set()
    for n_key, rec in (block.items() if isinstance(block, dict) else []):
        if not n_key.isdigit():
            continue
        if len(rec.get('r2_f0', [])) >= N_RUNS and len(rec.get('r2_spl', [])) >= N_RUNS:
            done.add(int(n_key))
    return done


def run_method(df, skip_existing=True):
    done = _existing_complete_ns('GP') if skip_existing else set()
    out = {}
    for n in N_TARGETS:
        if n in done:
            print(f"  N={n:>4}  skip (already in JSON)")
            continue
        runs = [run_single(n, df, random_state=42 + r) for r in range(N_RUNS)]
        r2_f0 = [r['r2_f0'] for r in runs]
        r2_spl = [r['r2_spl'] for r in runs]
        out[str(n)] = {'r2_f0': r2_f0, 'r2_spl': r2_spl,
                       'n_train': runs[0]['n_train']}
        print(f"  N={n:>4}  F0 R2 = {np.mean(r2_f0):+.3f} +/- {np.std(r2_f0):.3f}  "
              f"SPL R2 = {np.mean(r2_spl):+.3f} +/- {np.std(r2_spl):.3f}")
    return out


# ==================== I/O ====================
def load_tbcm():
    """Load TBCM dataset matching TBCM_TransferRF/TBCM_SmallData conventions."""
    p = os.path.join(script_dir, 'dataset_TBCM.csv')
    if not os.path.exists(p):
        raise FileNotFoundError(
            f"TBCM dataset not found at {p}. Ask Callum for the file or check "
            f"~/Downloads/ — TBCM_TransferRF.py:212 expects index_col=0."
        )
    df = pd.read_csv(p, index_col=0)
    if 'Ps' in df.columns:
        df = df.rename(columns={'Ps': 'PS'})
    df = df.dropna(subset=TARGETS).reset_index(drop=True)
    needed = SHARED_FEATURES + TARGETS
    missing = [c for c in needed if c not in df.columns]
    if missing:
        raise ValueError(f"TBCM CSV missing columns: {missing}. Has: {list(df.columns)}")
    return df[needed]


def merge_into_alternates(method_results, method_name='GP'):
    out_path = os.path.join(script_dir, 'results', 'alternates_results.json')
    os.makedirs(os.path.dirname(out_path), exist_ok=True)
    existing = {}
    if os.path.exists(out_path):
        with open(out_path) as f:
            existing = json.load(f)
    method_block = existing.get(method_name, {}) if isinstance(
        existing.get(method_name), dict) else {}
    for n_key, rec in method_results.items():
        method_block[n_key] = rec
    existing[method_name] = method_block
    all_n = sorted({int(k) for k in method_block.keys() if k.isdigit()})
    meta = existing.get('_meta', {}) or {}
    meta['n_targets'] = sorted(set(meta.get('n_targets', [])) | set(all_n))
    meta['n_runs'] = N_RUNS
    meta['shared_features'] = SHARED_FEATURES
    meta['targets'] = TARGETS
    meta['test_pool_size'] = TEST_POOL_SIZE
    meta['domain'] = 'TBCM'
    existing['_meta'] = meta
    with open(out_path, 'w') as f:
        json.dump(existing, f, indent=2)
    return out_path


def main():
    print("=" * 70)
    print("TBCM ALTERNATES — GP (Matern 2.5 + WhiteKernel)")
    print("=" * 70)
    df = load_tbcm()
    print(f"  rows: {len(df)}")
    print(f"  F0 range: [{df['F0'].min():.1f}, {df['F0'].max():.1f}]")
    print(f"  SPL range: [{df['SPL'].min():.1f}, {df['SPL'].max():.1f}]")
    print(f"  PS range:  [{df['PS'].min():.1f}, {df['PS'].max():.1f}]")
    results = run_method(df)
    out_path = merge_into_alternates(results, 'GP')
    print(f"\nResults written to: {out_path}")


if __name__ == '__main__':
    main()
```

- [ ] **Step 1: Create BM_CrossDomain.py**

```python
"""
Cross-domain alternates figure: GP/TabPFN vs best-transfer at each N, across
three target domains (BM, TBCM, Female BCM). Reads alternates_results.json
from each domain plus the matching transfer JSON or hardcoded small-N table.

Outputs:
  Beam_Membrane/figs/cross_domain_alternates.png  (3 panels side-by-side)
  Beam_Membrane/figs/cross_domain_table.png       (per-domain head-to-head)
"""

import json
import os
import numpy as np
import matplotlib.pyplot as plt

ROOT = os.path.abspath(os.path.join(os.path.dirname(__file__), '..'))
FIGS_DIR = os.path.join(os.path.dirname(__file__), 'figs')

DOMAINS = [
    {
        'name': 'BM',
        'title': 'Beam-Membrane (FEM)',
        'alt_json': os.path.join(ROOT, 'Beam_Membrane', 'results', 'alternates_results.json'),
        'transfer_table': {  # from BM_Showcase.SMALL_N_TRANSFER, best per N
            10: 0.075, 20: 0.054, 30: 0.187, 50: 0.193,
            75: 0.294, 100: 0.280, 200: 0.586, 500: 0.739,
        },
        'transfer_label': 'best BCM->BM transfer',
    },
    {
        'name': 'TBCM',
        'title': 'TBCM (triangular body-cover)',
        'alt_json': os.path.join(ROOT, 'TBCM', 'results', 'alternates_results.json'),
        'transfer_table': None,  # filled at runtime from rf_transfer_small_n.json
        # Small-N JSON produced by Task 5.5 (TBCM_SmallData.py). The original
        # rf_transfer_results.json is fraction-based (smallest n=1379) and does
        # NOT overlap with the N=5..500 alternates regime — do not use here.
        'transfer_json': os.path.join(ROOT, 'TBCM', 'results', 'rf_transfer_small_n.json'),
        'transfer_label': 'best BCM->TBCM transfer (TransRF)',
    },
    {
        'name': 'FemaleBCM',
        'title': 'Female BCM (Male -> Female transfer)',
        'alt_json': os.path.join(ROOT, 'VocalFoldRegression', 'BCM Model',
                                  'Alternates', 'results', 'alternates_results.json'),
        'transfer_table': None,
        # Per 2026-05-12 decision: RF transfer only as the comparator.
        # Existing artifact has r2_avg precomputed for NN/RF/PR at N=[25,50,100,250,500,843].
        'transfer_csv': os.path.join(ROOT, 'VocalFoldRegression', 'BCM Model',
                                      'ResgressorAnalysis', 'figs',
                                      'all_regressors_transfer_comparison.csv'),
        'transfer_label': 'Male->Female RF transfer',
    },
]

COLORS = {'GP': '#0f766e', 'TabPFN': '#d97706', 'transfer': '#475569'}


def load_alternates(path):
    if not os.path.exists(path):
        return {}
    with open(path) as f:
        raw = json.load(f)
    out = {}
    for method in ('GP', 'TabPFN'):
        if method not in raw:
            continue
        by_n = {}
        for n_str, rec in raw[method].items():
            if not n_str.isdigit():
                continue
            f0 = np.asarray(rec['r2_f0'])
            spl = np.asarray(rec['r2_spl'])
            by_n[int(n_str)] = (f0 + spl) / 2.0
        out[method] = by_n
    return out


def load_tbcm_transfer(path):
    """Pull TransRF mean avg-R^2 per N from the small-N TBCM JSON produced
    by Task 5.5 (TBCM_SmallData.py). Schema matches alternates_results.json:
        {'transrf': {'10': {'r2_f0': [...], 'r2_spl': [...], 'n_train': 10}, ...},
         'target_only': {...}, ...}
    """
    if not os.path.exists(path):
        return {}
    with open(path) as f:
        data = json.load(f)
    block = data.get('transrf', {})
    out = {}
    for n_str, rec in block.items():
        if not n_str.isdigit():
            continue
        f0 = np.asarray(rec['r2_f0'])
        spl = np.asarray(rec['r2_spl'])
        # mean over bootstrap runs, then avg of (f0, spl)
        out[int(n_str)] = float((f0.mean() + spl.mean()) / 2.0)
    return out


def load_female_transfer(csv_path):
    """Male->Female RF transfer comparator. Reads
    all_regressors_transfer_comparison.csv (schema:
        sample_size, regressor, r2_f0, r2_spl, r2_avg, mae_f0, mae_spl
    ), filters to regressor == 'RF' per the 2026-05-12 scope decision,
    and returns {sample_size: r2_avg}.
    """
    if not os.path.exists(csv_path):
        return {}
    import pandas as pd
    df = pd.read_csv(csv_path)
    required = {'sample_size', 'regressor', 'r2_avg'}
    if not required.issubset(df.columns):
        # Defensive: if schema changed, fall back to recomputing avg.
        if {'sample_size', 'regressor', 'r2_f0', 'r2_spl'}.issubset(df.columns):
            df = df.copy()
            df['r2_avg'] = (df['r2_f0'] + df['r2_spl']) / 2.0
        else:
            return {}
    rf = df[df['regressor'] == 'RF']
    return {int(row['sample_size']): float(row['r2_avg'])
            for _, row in rf.iterrows()}


def main():
    fig, axes = plt.subplots(1, 3, figsize=(16, 5.5), sharey=True)

    for ax, dom in zip(axes, DOMAINS):
        alt = load_alternates(dom['alt_json'])
        if not alt:
            ax.text(0.5, 0.5, f"{dom['name']}\n(no data yet)",
                    ha='center', va='center', transform=ax.transAxes,
                    fontsize=12, alpha=0.6)
            ax.set_title(dom['title'])
            continue

        for method, color in [('GP', COLORS['GP']), ('TabPFN', COLORS['TabPFN'])]:
            if method not in alt:
                continue
            ns = sorted(alt[method])
            means = np.array([alt[method][n].mean() for n in ns])
            stds = np.array([alt[method][n].std() for n in ns])
            ax.fill_between(ns, means - stds, means + stds, color=color, alpha=0.15)
            ax.plot(ns, means, color=color, lw=2.4, marker='o', markersize=6,
                    label=f'{method} (non-transfer)')

        # Transfer
        if dom['transfer_table']:
            xfer = dom['transfer_table']
        elif dom.get('transfer_json'):
            xfer = load_tbcm_transfer(dom['transfer_json'])
        elif dom.get('transfer_csv'):
            xfer = load_female_transfer(dom['transfer_csv'])
        else:
            xfer = {}
        if xfer:
            xs = sorted(xfer)
            ys = [xfer[n] for n in xs]
            ax.plot(xs, ys, color=COLORS['transfer'], lw=2.0, ls='--',
                    marker='s', markersize=5, label=dom['transfer_label'])

        ax.set_xscale('log')
        ax.set_xlabel('Number of target-domain training samples')
        ax.set_title(dom['title'], fontweight='bold')
        ax.axhline(0, color='black', lw=0.5, alpha=0.4)
        ax.set_ylim(-0.2, 1.0)
        ax.grid(True, which='both', alpha=0.25)
        ax.legend(loc='lower right', fontsize=9)

    axes[0].set_ylabel('Average R²  (F0 + SPL) / 2')
    fig.suptitle('GP / TabPFN vs transfer learning — three target domains',
                 fontweight='bold', fontsize=14, y=1.02)
    fig.tight_layout()
    out_path = os.path.join(FIGS_DIR, 'cross_domain_alternates.png')
    fig.savefig(out_path, dpi=180, bbox_inches='tight')
    plt.close(fig)
    print(f'wrote {out_path}')


if __name__ == '__main__':
    main()
```

## Task 7: Heatmaps (BM_Heatmaps.py)

**Files:**
- Create: `Beam_Membrane/BM_Heatmaps.py`

For each domain, train GP and TabPFN at N=50 (small enough that the data-efficiency story holds, large enough that the surfaces are well-defined). Build a 2D grid over `(a_CT, a_TA)` at fixed `PS = median`. Predict F0 over the grid with each method; render as a heatmap; overlay scattered training points colored by their true F0.

Layout: per domain, a 1×3 figure (GP, TabPFN, best-transfer-method) showing F0 surface. Repeat for SPL if desired (probably defer SPL panels — email volume).

- [ ] **Step 1: Create BM_Heatmaps.py skeleton**

```python
"""
Muscle-activation x F0 heatmaps comparing GP, TabPFN, and best transfer.
Answers Jesus's 2026-05-12 question: do the new methods replicate the
nonlinear (a_CT, a_TA) -> F0 trend, not just hit high R^2?

For each domain (BM, TBCM, female BCM):
  - Train GP and TabPFN at N=50 with random_state=42
  - Train the best transfer method at the same N (BM: TransRF, TBCM: TransRF,
    Female: NN transfer if available — else skip the third panel)
  - Sweep (a_CT, a_TA) on a 50x50 grid at PS = dataset median
  - Predict F0; render heatmaps side-by-side
  - Overlay the 50 training points as scatter, colored by true F0
  - Save figs/heatmap_<domain>_F0.png
"""

import json
import os
import warnings
warnings.filterwarnings('ignore')

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern, WhiteKernel, ConstantKernel as C
from sklearn.preprocessing import StandardScaler

ROOT = os.path.abspath(os.path.join(os.path.dirname(__file__), '..'))
FIGS_DIR = os.path.join(os.path.dirname(__file__), 'figs')
GRID_RES = 50
N_TRAIN = 50
RANDOM_STATE = 42
TARGET = 'F0'


def fit_gp_one_target(X_train_sc, y_train_sc):
    kernel = (C(1.0, (1e-3, 1e3))
              * Matern(length_scale=1.0, length_scale_bounds=(1e-2, 1e2), nu=2.5)
              + WhiteKernel(noise_level=1e-2, noise_level_bounds=(1e-5, 1e1)))
    gp = GaussianProcessRegressor(kernel=kernel, normalize_y=False,
                                  n_restarts_optimizer=3, random_state=42)
    gp.fit(X_train_sc, y_train_sc)
    return gp


def fit_tabpfn_one_target(X_train_sc, y_train_sc):
    from tabpfn_client import TabPFNRegressor
    r = TabPFNRegressor(random_state=42)
    r.fit(X_train_sc, y_train_sc)
    return r


def build_grid(df, ps_value):
    a_ct = np.linspace(df['a_CT'].min(), df['a_CT'].max(), GRID_RES)
    a_ta = np.linspace(df['a_TA'].min(), df['a_TA'].max(), GRID_RES)
    AC, AT = np.meshgrid(a_ct, a_ta)
    grid_X = np.column_stack([AC.ravel(), AT.ravel(),
                              np.full(AC.size, ps_value)])
    return AC, AT, grid_X


def heatmap_for_domain(df, name, transfer_pred_fn=None, transfer_label=None):
    df = df.sample(n=min(len(df), 2000), random_state=RANDOM_STATE)
    df_train = df.sample(n=N_TRAIN, random_state=RANDOM_STATE)
    X_train = df_train[['a_CT', 'a_TA', 'PS']].values
    y_train = df_train[TARGET].values

    scaler_X = StandardScaler().fit(X_train)
    scaler_y = StandardScaler().fit(y_train.reshape(-1, 1))
    X_train_sc = scaler_X.transform(X_train)
    y_train_sc = scaler_y.transform(y_train.reshape(-1, 1)).ravel()

    ps_value = float(df['PS'].median())
    AC, AT, grid_X = build_grid(df, ps_value)
    grid_sc = scaler_X.transform(grid_X)

    gp = fit_gp_one_target(X_train_sc, y_train_sc)
    pred_gp_sc = gp.predict(grid_sc)
    pred_gp = scaler_y.inverse_transform(pred_gp_sc.reshape(-1, 1)).ravel()

    try:
        tab = fit_tabpfn_one_target(X_train_sc, y_train_sc)
        pred_tab_sc = tab.predict(grid_sc)
        pred_tab = scaler_y.inverse_transform(pred_tab_sc.reshape(-1, 1)).ravel()
        have_tab = True
    except Exception as e:
        print(f'  {name}: TabPFN failed ({e!r}); rendering only GP and transfer.')
        pred_tab = None
        have_tab = False

    if transfer_pred_fn is not None:
        try:
            pred_xfer = transfer_pred_fn(X_train, y_train, grid_X)
            have_xfer = True
        except Exception as e:
            print(f'  {name}: transfer baseline failed ({e!r}); skipping panel.')
            pred_xfer = None
            have_xfer = False
    else:
        pred_xfer = None
        have_xfer = False

    panels = [('GP', pred_gp.reshape(GRID_RES, GRID_RES))]
    if have_tab:
        panels.append(('TabPFN', pred_tab.reshape(GRID_RES, GRID_RES)))
    if have_xfer:
        panels.append((transfer_label or 'transfer',
                       pred_xfer.reshape(GRID_RES, GRID_RES)))

    vmin = min(p.min() for _, p in panels)
    vmax = max(p.max() for _, p in panels)
    vmin = min(vmin, y_train.min())
    vmax = max(vmax, y_train.max())

    fig, axes = plt.subplots(1, len(panels),
                              figsize=(4.8 * len(panels), 4.5), sharey=True)
    if len(panels) == 1:
        axes = [axes]
    for ax, (label, surf) in zip(axes, panels):
        im = ax.pcolormesh(AC, AT, surf, cmap='viridis',
                           vmin=vmin, vmax=vmax, shading='auto')
        ax.scatter(df_train['a_CT'], df_train['a_TA'],
                   c=df_train[TARGET], cmap='viridis',
                   vmin=vmin, vmax=vmax, edgecolor='white', s=40, lw=0.8,
                   label='training points (true F0)')
        ax.set_xlabel('a_CT')
        ax.set_title(f'{name}: {label}  (PS={ps_value:.0f})', fontweight='bold')
        if ax is axes[0]:
            ax.set_ylabel('a_TA')
    fig.colorbar(im, ax=axes, label='F0 (Hz)', shrink=0.85)
    fig.suptitle(f'F0 surface — {name}, N={N_TRAIN} training points',
                 fontweight='bold', y=1.02)

    out_path = os.path.join(FIGS_DIR, f'heatmap_{name}_F0.png')
    fig.savefig(out_path, dpi=180, bbox_inches='tight')
    plt.close(fig)
    print(f'  wrote {out_path}')


def transfer_pred_transrf(source_df_path, X_train, y_train, grid_X):
    """TransRF baseline: BCM source RF + target-domain RF on residuals + augmented.
    Lightweight reimplementation — predicts F0 only (one-target). Returns
    grid predictions in original scale.
    """
    from sklearn.ensemble import RandomForestRegressor
    src = pd.read_csv(source_df_path)
    if 'Ps' in src.columns:
        src = src.rename(columns={'Ps': 'PS'})
    src = src.dropna(subset=['F0', 'SPL'])[['a_CT', 'a_TA', 'PS', 'F0']]
    X_src = src[['a_CT', 'a_TA', 'PS']].values
    y_src = src['F0'].values

    scaler_X_src = StandardScaler().fit(X_src)
    rf_src = RandomForestRegressor(n_estimators=100, max_depth=10,
                                    random_state=42, n_jobs=-1)
    rf_src.fit(scaler_X_src.transform(X_src), y_src)

    pred_src_train = rf_src.predict(scaler_X_src.transform(X_train))
    residual = y_train - pred_src_train
    rf_res = RandomForestRegressor(n_estimators=20, max_depth=3,
                                    min_samples_leaf=5, random_state=42)
    rf_res.fit(X_train, residual)

    pred_src_grid = rf_src.predict(scaler_X_src.transform(grid_X))
    pred_res_grid = rf_res.predict(grid_X)
    return pred_src_grid + pred_res_grid


def main():
    print('Building cross-domain F0 heatmaps...')
    os.makedirs(FIGS_DIR, exist_ok=True)

    # BM
    bm = pd.read_csv(os.path.join(ROOT, 'Beam_Membrane', 'dataset_BM.csv'))
    if 'Ps' in bm.columns:
        bm = bm.rename(columns={'Ps': 'PS'})
    bm = bm.dropna(subset=['F0', 'SPL'])[['a_CT', 'a_TA', 'PS', 'F0', 'SPL']]
    bcm_path = os.path.join(ROOT, 'VocalFoldRegression', 'BCM Model', 'MaleBCM.csv')

    def bm_xfer(X_train, y_train, grid_X):
        return transfer_pred_transrf(bcm_path, X_train, y_train, grid_X)

    heatmap_for_domain(bm, 'BM', transfer_pred_fn=bm_xfer,
                       transfer_label='best BCM->BM transfer (TransRF)')

    # TBCM
    tbcm_path = os.path.join(ROOT, 'TBCM', 'dataset_TBCM.csv')
    if os.path.exists(tbcm_path):
        tbcm = pd.read_csv(tbcm_path, index_col=0)
        if 'Ps' in tbcm.columns:
            tbcm = tbcm.rename(columns={'Ps': 'PS'})
        tbcm = tbcm.dropna(subset=['F0', 'SPL'])[['a_CT', 'a_TA', 'PS', 'F0', 'SPL']]
        def tbcm_xfer(X_train, y_train, grid_X):
            return transfer_pred_transrf(bcm_path, X_train, y_train, grid_X)
        heatmap_for_domain(tbcm, 'TBCM', transfer_pred_fn=tbcm_xfer,
                           transfer_label='best BCM->TBCM transfer (TransRF)')
    else:
        print('  TBCM: dataset_TBCM.csv not found — skipping')

    # Female BCM
    fem = pd.read_csv(os.path.join(ROOT, 'VocalFoldRegression', 'BCM Model',
                                    'FemaleBCM.csv'))
    if 'Ps' in fem.columns and 'PS' not in fem.columns:
        fem = fem.rename(columns={'Ps': 'PS'})
    fem = fem.dropna(subset=['F0', 'SPL'])[['a_CT', 'a_TA', 'PS', 'F0', 'SPL']]
    def fem_xfer(X_train, y_train, grid_X):
        return transfer_pred_transrf(bcm_path, X_train, y_train, grid_X)
    heatmap_for_domain(fem, 'FemaleBCM', transfer_pred_fn=fem_xfer,
                       transfer_label='best Male->Female transfer (RF-residual)')
```
